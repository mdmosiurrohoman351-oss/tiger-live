Admin নিয়োগ: TIGER ID দিয়ে CREATE OR REPLACE FUNCTION public.tiger_live_owner_assign_admin( p_tiger_id text ) RETURNS uuid LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $$ DECLARE v_user_id uuid; v_role text; v_count integer; v_permission text; BEGIN IF NOT public.is_tiger_live_owner() THEN RAISE EXCEPTION 'Only Super Master Admin can assign admins'; END IF; SELECT count(*) INTO v_count FROM public.profiles WHERE upper(trim(tiger_id)) = upper(trim(p_tiger_id)); IF v_count = 0 THEN RAISE EXCEPTION 'TIGER ID not found'; ELSIF v_count > 1 THEN RAISE EXCEPTION 'Duplicate TIGER ID found'; END IF; SELECT id INTO v_user_id FROM public.profiles WHERE upper(trim(tiger_id)) = upper(trim(p_tiger_id)); SELECT role INTO v_role FROM public.tiger_live_roles WHERE user_id = v_user_id FOR UPDATE; IF v_role = 'owner' THEN RAISE EXCEPTION 'Owner cannot be assigned as admin'; ELSIF v_role = 'moderator' THEN RAISE EXCEPTION 'This user is already a moderator'; ELSIF v_role IS NULL THEN INSERT INTO public.tiger_live_roles (user_id, role) VALUES (v_user_id, 'admin') ON CONFLICT (user_id) DO NOTHING; END IF; SELECT role INTO v_role FROM public.tiger_live_roles WHERE user_id = v_user_id; IF v_role <> 'admin' THEN RAISE EXCEPTION 'Could not assign admin role'; END IF; FOREACH v_permission IN ARRAY ARRAY[ 'user_management', 'agency_management', 'host_management', 'gift_management', 'diamond_management', 'wallet_recharge', 'vip_management', 'audio_video_room', 'support_management', 'admin_management', 'user_ban', 'reports_statistics' ] LOOP INSERT INTO public.tiger_live_admin_permissions (user_id, permission, is_enabled) VALUES (v_user_id, v_permission, false) ON CONFLICT (user_id, permission) DO NOTHING; END LOOP; RETURN v_user_id; END; $$; -- 2. Admin-এর Permission ON/OFF CREATE OR REPLACE FUNCTION public.tiger_live_owner_set_permission( p_user_id uuid, p_permission text, p_enabled boolean ) RETURNS boolean LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $$ BEGIN IF NOT public.is_tiger_live_owner() THEN RAISE EXCEPTION 'Only Super Master Admin can change permissions'; END IF; IF p_permission NOT IN ( 'user_management', 'agency_management', 'host_management', 'gift_management', 'diamond_management', 'wallet_recharge', 'vip_management', 'audio_video_room', 'support_management', 'admin_management', 'user_ban', 'reports_statistics' ) THEN RAISE EXCEPTION 'Invalid permission'; END IF; IF NOT EXISTS ( SELECT 1 FROM public.tiger_live_roles WHERE user_id = p_user_id AND role = 'admin' ) THEN RAISE EXCEPTION 'Target user is not an admin'; END IF; INSERT INTO public.tiger_live_admin_permissions (user_id, permission, is_enabled) VALUES (p_user_id, p_permission, p_enabled) ON CONFLICT (user_id, permission) DO UPDATE SET is_enabled = EXCLUDED.is_enabled, updated_at = now(); RETURN true; END; $$; -- 3. Admin তালিকা দেখা CREATE OR REPLACE FUNCTION public.tiger_live_owner_list_admins() RETURNS TABLE ( user_id uuid, tiger_id text, username text, created_at timestamptz ) LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $$ BEGIN IF NOT public.is_tiger_live_owner() THEN RAISE EXCEPTION 'Only Super Master Admin can view admins'; END IF; RETURN QUERY SELECT r.user_id, p.tiger_id, p.username, r.created_at FROM public.tiger_live_roles r JOIN public.profiles p ON p.id = r.user_id WHERE r.role = 'admin' ORDER BY r.created_at DESC; END; $$; -- 4. Admin অপসারণ CREATE OR REPLACE FUNCTION public.tiger_live_owner_remove_admin( p_user_id uuid ) RETURNS boolean LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $$ DECLARE v_role text; BEGIN IF NOT public.is_tiger_live_owner() THEN RAISE EXCEPTION 'Only Super Master Admin can remove admins'; END IF; SELECT role INTO v_role FROM public.tiger_live_roles WHERE user_id = p_user_id FOR UPDATE; IF v_role IS DISTINCT FROM 'admin' THEN RAISE EXCEPTION 'Target user is not an admin'; END IF; DELETE FROM public.tiger_live_admin_permissions WHERE user_id = p_user_id; DELETE FROM public.tiger_live_roles WHERE user_id = p_user_id AND role = 'admin'; RETURN true; END; $$; -- 5. নিরাপদ RPC অনুমতি REVOKE ALL ON FUNCTION public.tiger_live_owner_assign_admin(text) FROM PUBLIC, anon, authenticated; GRANT EXECUTE ON FUNCTION public.tiger_live_owner_assign_admin(text) TO authenticated; REVOKE ALL ON FUNCTION public.tiger_live_owner_set_permission(uuid, text, boolean) FROM PUBLIC, anon, authenticated; GRANT EXECUTE ON FUNCTION public.tiger_live_owner_set_permission(uuid, text, boolean) TO authenticated; REVOKE ALL ON FUNCTION public.tiger_live_owner_list_admins() FROM PUBLIC, anon, authenticated; GRANT EXECUTE ON FUNCTION public.tiger_live_owner_list_admins() TO authenticated; REVOKE ALL ON FUNCTION public.tiger_live_owner_remove_admin(uuid) FROM PUBLIC, anon, authenticated; GRANT EXECUTE ON FUNCTION public.tiger_live_owner_remove_admin(uuid) TO authenticated;
